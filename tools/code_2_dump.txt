Problem Statement :   
Write a program to store the first-year percentage of students in an array. Write function for 
sorting array of floating-point numbers in ascending order using-Selection Sort Bubble sort 
and display top five scores. 
 
Input code: 
#include <stdio.h> 
#include <stdlib.h> 
#define MAX_STUD 100 
 
int main() 
{ 
    float percentage[MAX_STUD]; 
    int num = 0, ch, i, j; 
    float temp; 
 
    do 
    { 
        printf("__Student percentage management system__\n"); 
        printf("1.Enter student percentage\n2.Selection sort\n3.Bubble sort\n4.Top 5\n5.Display 
all student percentage\n6.exit\n"); 
        printf("Enter your choice:"); 
        scanf("%d",&ch); 
        switch(ch) 
        { 
            case 1: 
                printf("Enter number of students (max %d): ", MAX_STUD); 
                scanf("%d", &num); 
                if(num > MAX_STUD || num <= 0) 
/* --- PAGE 2 --- */
                { 
                    printf("Insert a valid count!\n"); 
                    num = 0; 
                } 
                else 
                { 
                    printf("Enter percentages of students:\n"); 
                    for(i = 0; i < num; i++) 
                    { 
                        printf("Percentage of student %d: ", i + 1); 
                        scanf("%f", &percentage[i]); 
                    } 
                } 
                break; 
 
            case 2: 
                if(num == 0) 
                { 
                    printf("No student data available! Please enter data first.\n"); 
                    break; 
                } 
                for(i = 0; i < num - 1; i++) 
                { 
                    int min = i; 
                    for(j = i + 1; j < num; j++) 
                    { 
                        if(percentage[j] < percentage[min]) 
                        { 
                            min = j; 
/* --- PAGE 3 --- */
                        } 
                    } 
                    if(min != i) 
                    { 
                        temp = percentage[i]; 
                        percentage[i] = percentage[min]; 
                        percentage[min] = temp; 
                    } 
                } 
                printf("\nPercentages sorted (Ascending) using Selection Sort:\n"); 
                for(i = 0; i < num; i++) 
                { 
                    printf("%.2f%%\n", percentage[i]); 
                } 
                break; 
 
            case 3: 
                if(num == 0) 
                { 
                    printf("No student data available! Please enter data first.\n"); 
                    break; 
                } 
                for(i = 0; i < num - 1; i++) 
                { 
                    for(j = 0; j < num - 1 - i; j++) 
                    { 
                        if(percentage[j] > percentage[j + 1]) 
                        { 
                            temp = percentage[j]; 
/* --- PAGE 4 --- */
                            percentage[j] = percentage[j + 1]; 
                            percentage[j + 1] = temp; 
                        } 
                    } 
                } 
                printf("\nPercentages sorted (Ascending) using Bubble Sort:\n"); 
                for(i = 0; i < num; i++) 
                { 
                    printf("%.2f%%\n", percentage[i]); 
                } 
                break; 
 
            case 4: 
                if(num == 0) 
                { 
                    printf("No student data available! Please enter data first.\n"); 
                    break; 
                } 
 
                for(i = 0; i < num - 1; i++) 
                { 
                    int min = i; 
                    for(j = i + 1; j < num; j++) 
                    { 
                        if(percentage[j] < percentage[min]) 
                        { 
                            min = j; 
                        } 
                    } 
/* --- PAGE 5 --- */
                    if(min != i) 
                    { 
                        temp = percentage[i]; 
                        percentage[i] = percentage[min]; 
                        percentage[min] = temp; 
                    } 
                } 
 
                printf("\nTop Percentages:\n"); 
                int count = 0; 
                for(i = num - 1; i >= 0 && count < 5; i--) 
                { 
                    printf("Rank %d: %.2f%%\n", count + 1, percentage[i]); 
                    count++; 
                } 
                break; 
 
            case 5: 
                if(num == 0) 
                { 
                    printf("No student data available! Please enter data first.\n"); 
                } 
                else 
                { 
                    printf("\nAll Student Percentages:\n"); 
                    for(i = 0; i < num; i++) 
                    { 
                        printf("Student %d: %.2f%%\n", i + 1, percentage[i]); 
                    } 
/* --- PAGE 6 --- */
                } 
                break; 
 
            case 6: 
                printf("Exiting program...\n"); 
                break; 
 
            default: 
                printf("Invalid choice! Please enter a option between 1 and 6.\n"); 
                break; 
        } 
    } while(ch != 6); 
 
    return 0; 
}
