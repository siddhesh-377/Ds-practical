b) Write a program to store roll numbers of student array who attended training

attended a training program or not, using Binary search.

void linearSearch(int a[], int n, int key) 
{ 
    int i, found = 0; 
 
    for (i = 0; i < n; i++)  
{ 
        if (a[i] == key)  
       { 
            printf("Element found at index %d (position %d)\n" , i, i + 1); 
            found = 1; 
            break; 
        } 
    } 
 if (!found) 
     {

} 
void binarySearch(int a[], int n, int key) 
 { 
    int low = 0, high = n - 1, mid; 
    int found = 0; 
 
    while (low <= high) 
    { 
        mid = low + (high - low) / 2; 
 
        if (a[mid] == key) 
        { 
            printf("Element found at index %d (position %d)\n" , mid, mid + 1); 
            found = 1; 
            break; 
        } else if (a[mid] < key) 
         { 
            low = mid + 1; 
        } else { 
            high = mid - 1; 
        } 
    } 
 
    if (!found) { 
        printf("Element not found in the array!\n"); 
    } 
/* --- PAGE 5 --- */
} 
void sortArray(int a[], int n)  
{ 
    int i, j, temp; 
    for (i = 0; i < n - 1; i++) { 
        for (j = 0; j < n - i - 1; j++) 
       { 
            if (a[j] > a[j + 1]) { 
                temp = a[j]; 
                a[j] = a[j + 1]; 
                a[j + 1] = temp; 
            } 
        } 
    } 
}
